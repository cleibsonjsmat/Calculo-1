\begin{questao}[1,5 ponto]{Operações Fundamentais}
Sejam os números complexos $z_1 = 3 + 4i$ e $z_2 = 1 - 2i$. Efetue as seguintes operações:
\begin{enumerate}[label=\alph*)]
    \item $z_1 + 2z_2$
    \item $z_1 \cdot z_2$
    \item $\dfrac{z_1}{z_2}$
\end{enumerate}
\end{questao}

% Espaço para o aluno responder caso seja impressa
\espacoresposta{3cm}

\begin{questao}[2,0 pontos]{Potências da Unidade Imaginária}
Determine o valor numérico exato da seguinte expressão algébrica:
\[
S = i^{2026} + i^{1999} - 3i^{47}
\]
\end{questao}

% Exemplo de questão acompanhada de Gabarito/Resolução
\begin{gabarito}
\textbf{Passo 1:} Dividir os expoentes por $4$ e determinar os restos:
\begin{itemize}[noitemsep]
    \item $2026 = 4(506) + 2 \implies i^{2026} = i^2 = -1$
    \item $1999 = 4(499) + 3 \implies i^{1999} = i^3 = -i$
    \item $47 = 4(11) + 3 \implies i^{47} = i^3 = -i$
\end{itemize}
\textbf{Passo 2:} Substituir na expressão original:
\[
S = (-1) + (-i) - 3(-i) = -1 - i + 3i = -1 + 2i
\]
\end{gabarito}
