\item Determine o domínio das funções:
\begin{enumerate}
    \item $f(x) = \dfrac{\sqrt{x+2}}{x-1}$
    \item $g(x) = \ln(4 - x^2) + \dfrac{1}{|x|}$
    \item $h(x) = \begin{cases} 
      \dfrac{1}{x} & \text{se } x < 0 \\ 
      \sqrt{x} & \text{se } x \geq 0 
    \end{cases}$
\end{enumerate}

\item Para cada função abaixo, encontre a imagem:
\begin{enumerate}
    \item $f(x) = 3 - 2x^2$
    \item $g(x) = 2 + \sqrt{x - 1}$
    \item $h(x) = \lceil x \rceil$ (função teto) para $x \in [-1, 2]$
\end{enumerate}



\item Verifique se as funções são bijetoras e, quando possível, encontre $f^{-1}(x)$:
\begin{enumerate}
    \item $f(x) = \dfrac{2x + 3}{5}$
    \item $g(x) = x^2 - 4x$ (para $x \geq 2$)
    \item $h(x) = e^{3x-1}$
\end{enumerate}

\item A função $T(C) = \dfrac{9}{5}C + 32$ converte Celsius para Fahrenheit. Encontre:
\begin{enumerate}
    \item $T^{-1}(F)$
    \item O valor de $C$ quando $T(C) = 100^\circ F$
\end{enumerate}


\item Dadas $f(x) = \sqrt{x-1}$ e $g(x) = x^2 + 2$, calcule:

\begin{enumerate}[a)]
    \item $(f \circ g)(3)$
    \item $(g \circ f)(5)$
    \item $(f \circ f)(10)$
    \item O domínio de $f \circ g$
\end{enumerate}

\item Se $h(x) = (x^3 + 1)^2$, encontre funções $f$ e $g$ tais que $h = f \circ g$.

\end{enumerate}
